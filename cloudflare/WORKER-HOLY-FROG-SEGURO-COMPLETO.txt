const CORS_HEADERS = {
  "Access-Control-Allow-Origin": "*",
  "Access-Control-Allow-Methods": "GET, POST, OPTIONS",
  "Access-Control-Allow-Headers": "Content-Type, Authorization",
};

const ADMIN_SESSION_SECONDS = 2 * 60 * 60;
const DAY_MS = 24 * 60 * 60 * 1000;
const LIFETIME_DAYS = 100000;

// Proteção complementar de login. Em Workers, este Map é apenas uma camada extra
// por isolate; para rate limit global, use também as regras de Rate Limiting do Cloudflare.
const ADMIN_LOGIN_WINDOW_MS = 15 * 60 * 1000;
const ADMIN_LOGIN_MAX_ATTEMPTS = 5;
const adminLoginAttempts = new Map();

function clientIp(request) {
  return text(request.headers.get("CF-Connecting-IP") || "unknown");
}

function normalizedOrigin(value) {
  return text(value).replace(/\/+$/, "").toLowerCase();
}

function allowedAdminOrigins(env) {
  return text(env.ADMIN_ALLOWED_ORIGINS)
    .split(",")
    .map(normalizedOrigin)
    .filter(Boolean);
}

function isAdminOriginAllowed(request, env) {
  const origin = normalizedOrigin(request.headers.get("Origin"));
  if (!origin) return true; // permite acesso direto/curl; autenticação continua obrigatória

  const allowed = allowedAdminOrigins(env);
  if (!allowed.length) return true; // compatibilidade até ADMIN_ALLOWED_ORIGINS ser configurado
  return allowed.includes(origin);
}

function adminRateState(request) {
  const key = clientIp(request);
  const now = Date.now();
  let state = adminLoginAttempts.get(key);

  if (!state || now - state.firstAt > ADMIN_LOGIN_WINDOW_MS) {
    state = { firstAt: now, count: 0, blockedUntil: 0 };
    adminLoginAttempts.set(key, state);
  }

  return { key, state, now };
}

function isAdminLoginBlocked(request) {
  const { state, now } = adminRateState(request);
  return Number(state.blockedUntil || 0) > now;
}

function registerAdminLoginFailure(request) {
  const { key, state, now } = adminRateState(request);
  state.count += 1;
  if (state.count >= ADMIN_LOGIN_MAX_ATTEMPTS) {
    state.blockedUntil = now + ADMIN_LOGIN_WINDOW_MS;
  }
  adminLoginAttempts.set(key, state);
}

function clearAdminLoginFailures(request) {
  adminLoginAttempts.delete(clientIp(request));
}

function json(data, status = 200) {
  return new Response(JSON.stringify(data), {
    status,
    headers: {
      ...CORS_HEADERS,
      "Content-Type": "application/json; charset=utf-8",
    },
  });
}

function text(value) {
  return String(value ?? "").trim();
}

function digits(value) {
  return String(value ?? "").replace(/\D/g, "");
}

function isEmail(value) {
  return /^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(String(value ?? "").trim());
}

/* =========================================================
   BASE64 / CRYPTO
========================================================= */

function base64UrlEncodeBytes(bytes) {
  let binary = "";
  for (const b of bytes) binary += String.fromCharCode(b);

  return btoa(binary)
    .replace(/\+/g, "-")
    .replace(/\//g, "_")
    .replace(/=+$/g, "");
}

function base64UrlEncodeText(value) {
  return base64UrlEncodeBytes(new TextEncoder().encode(value));
}

function base64UrlDecode(value) {
  value = String(value).replace(/-/g, "+").replace(/_/g, "/");
  while (value.length % 4) value += "=";

  const binary = atob(value);
  return Uint8Array.from(binary, c => c.charCodeAt(0));
}

function normalizePrivateKey(privateKey) {
  return String(privateKey || "")
    .trim()
    .replace(/^["']|["']$/g, "")
    .replace(/\\n/g, "\n");
}

function pemToArrayBuffer(pem) {
  const clean = normalizePrivateKey(pem)
    .replace("-----BEGIN PRIVATE KEY-----", "")
    .replace("-----END PRIVATE KEY-----", "")
    .replace(/\s/g, "");

  if (!clean) {
    throw new Error("FIREBASE_PRIVATE_KEY vazia.");
  }

  const binary = atob(clean);
  const bytes = new Uint8Array(binary.length);

  for (let i = 0; i < binary.length; i++) {
    bytes[i] = binary.charCodeAt(i);
  }

  return bytes.buffer;
}

async function adminHmacKey(env) {
  const sessionSecret = text(env.ADMIN_SESSION_SECRET) || text(env.ADMIN_PASSWORD);

  if (!sessionSecret) {
    throw new Error("ADMIN_SESSION_SECRET não configurado.");
  }

  return crypto.subtle.importKey(
    "raw",
    new TextEncoder().encode(sessionSecret),
    { name: "HMAC", hash: "SHA-256" },
    false,
    ["sign", "verify"]
  );
}

async function createAdminToken(env, email = "") {
  const now = Math.floor(Date.now() / 1000);

  const payload = {
    admin: true,
    email: text(email).toLowerCase(),
    iat: now,
    exp: now + ADMIN_SESSION_SECONDS,
  };

  const payload64 = base64UrlEncodeText(JSON.stringify(payload));
  const key = await adminHmacKey(env);

  const signature = await crypto.subtle.sign(
    "HMAC",
    key,
    new TextEncoder().encode(payload64)
  );

  return `${payload64}.${base64UrlEncodeBytes(new Uint8Array(signature))}`;
}

async function validateAdminToken(request, env) {
  try {
    const header = request.headers.get("Authorization") || "";
    if (!header.startsWith("Bearer ")) return false;

    const token = header.slice(7).trim();
    const parts = token.split(".");
    if (parts.length !== 2) return false;

    const [payload64, signature64] = parts;
    const key = await adminHmacKey(env);

    const ok = await crypto.subtle.verify(
      "HMAC",
      key,
      base64UrlDecode(signature64),
      new TextEncoder().encode(payload64)
    );

    if (!ok) return false;

    const payload = JSON.parse(
      new TextDecoder().decode(base64UrlDecode(payload64))
    );

    const now = Math.floor(Date.now() / 1000);

    return payload.admin === true && Number(payload.exp || 0) > now;
  } catch (_) {
    return false;
  }
}

async function requireAdmin(request, env) {
  return await validateAdminToken(request, env);
}

function readAdminTokenPayload(request) {
  try {
    const header = request.headers.get("Authorization") || "";
    if (!header.startsWith("Bearer ")) return null;
    const token = header.slice(7).trim();
    const [payload64] = token.split(".");
    if (!payload64) return null;
    return JSON.parse(new TextDecoder().decode(base64UrlDecode(payload64)));
  } catch (_) {
    return null;
  }
}

async function getGoogleAccessToken(env) {
  const projectId = text(env.FIREBASE_PROJECT_ID);
  const clientEmail = text(env.FIREBASE_CLIENT_EMAIL);
  const privateKey = normalizePrivateKey(env.FIREBASE_PRIVATE_KEY);

  if (!projectId) throw new Error("FIREBASE_PROJECT_ID não configurado.");
  if (!clientEmail) throw new Error("FIREBASE_CLIENT_EMAIL não configurado.");
  if (!privateKey) throw new Error("FIREBASE_PRIVATE_KEY não configurado.");

  const now = Math.floor(Date.now() / 1000);

  const header64 = base64UrlEncodeText(JSON.stringify({
    alg: "RS256",
    typ: "JWT",
  }));

  const payload64 = base64UrlEncodeText(JSON.stringify({
    iss: clientEmail,
    scope: "https://www.googleapis.com/auth/cloud-platform",
    aud: "https://oauth2.googleapis.com/token",
    iat: now,
    exp: now + 3600,
  }));

  const unsigned = `${header64}.${payload64}`;

  let cryptoKey;
  try {
    cryptoKey = await crypto.subtle.importKey(
      "pkcs8",
      pemToArrayBuffer(privateKey),
      { name: "RSASSA-PKCS1-v1_5", hash: "SHA-256" },
      false,
      ["sign"]
    );
  } catch (e) {
    console.error("PRIVATE KEY IMPORT ERROR", e);
    throw new Error("FIREBASE_PRIVATE_KEY inválida ou em formato incorreto.");
  }

  const signature = await crypto.subtle.sign(
    "RSASSA-PKCS1-v1_5",
    cryptoKey,
    new TextEncoder().encode(unsigned)
  );

  const jwt = `${unsigned}.${base64UrlEncodeBytes(new Uint8Array(signature))}`;

  const response = await fetch("https://oauth2.googleapis.com/token", {
    method: "POST",
    headers: {
      "Content-Type": "application/x-www-form-urlencoded",
    },
    body: new URLSearchParams({
      grant_type: "urn:ietf:params:oauth:grant-type:jwt-bearer",
      assertion: jwt,
    }),
  });

  const data = await response.json();

  if (!response.ok || !data.access_token) {
    console.error("GOOGLE OAUTH ERROR", JSON.stringify(data));
    throw new Error(
      data.error_description ||
      data.error ||
      "Não foi possível autenticar a conta de serviço."
    );
  }

  return data.access_token;
}

function firebaseAuthBase(env) {
  const projectId = encodeURIComponent(text(env.FIREBASE_PROJECT_ID));
  return `https://identitytoolkit.googleapis.com/v1/projects/${projectId}`;
}

async function authLookupByUid(env, token, uid) {
  const response = await fetch(`${firebaseAuthBase(env)}/accounts:lookup`, {
    method: "POST",
    headers: {
      Authorization: `Bearer ${token}`,
      "Content-Type": "application/json",
    },
    body: JSON.stringify({ localId: [uid] }),
  });

  const data = await response.json();

  if (!response.ok) {
    throw new Error(data?.error?.message || "Falha ao localizar usuário no Authentication.");
  }

  return data.users?.[0] || null;
}


async function authLookupByEmail(env, token, email) {
  const cleanEmail = text(email).toLowerCase();
  if (!cleanEmail) return null;

  const response = await fetch(`${firebaseAuthBase(env)}/accounts:lookup`, {
    method: "POST",
    headers: {
      Authorization: `Bearer ${token}`,
      "Content-Type": "application/json",
    },
    body: JSON.stringify({ email: [cleanEmail] }),
  });

  const data = await response.json();

  if (!response.ok) {
    throw new Error(data?.error?.message || "Falha ao localizar e-mail no Authentication.");
  }

  return data.users?.[0] || null;
}

async function authCreateUser(env, token, email, password) {
  const cleanEmail = text(email).toLowerCase();
  const cleanPassword = String(password || "");

  if (!isEmail(cleanEmail)) {
    throw new Error("A conta precisa ter um e-mail válido para ser recuperada.");
  }

  if (cleanPassword.length < 6) {
    throw new Error("A nova senha precisa ter pelo menos 6 caracteres.");
  }

  const projectId = encodeURIComponent(text(env.FIREBASE_PROJECT_ID));

  const response = await fetch(
    `https://identitytoolkit.googleapis.com/v1/projects/${projectId}/accounts`,
    {
      method: "POST",
      headers: {
        Authorization: `Bearer ${token}`,
        "Content-Type": "application/json",
      },
      body: JSON.stringify({
        email: cleanEmail,
        password: cleanPassword,
        emailVerified: false,
        disabled: false,
      }),
    }
  );

  const data = await response.json();

  if (!response.ok) {
    throw new Error(data?.error?.message || "Falha ao recriar usuário no Authentication.");
  }

  return data;
}

async function garantirAuthDaConta(env, token, user, whatsapp, novaSenha) {
  const email = text(user?.email).toLowerCase();
  const uidSalvo = text(user?.authUid);

  let authUser = null;

  if (uidSalvo) {
    try {
      authUser = await authLookupByUid(env, token, uidSalvo);
    } catch (_) {
      authUser = null;
    }
  }

  // Se o UID salvo estiver quebrado, tenta localizar pelo e-mail atual.
  if (!authUser && email) {
    try {
      authUser = await authLookupByEmail(env, token, email);
    } catch (_) {
      authUser = null;
    }
  }

  // Conta realmente antiga: existe no Firestore, mas nunca existiu (ou já não existe)
  // no Firebase Authentication. Recriamos sem mexer no plano/vencimento.
  if (!authUser) {
    const criado = await authCreateUser(env, token, email, novaSenha);
    const novoUid = text(criado.localId);

    if (!novoUid) {
      throw new Error("Authentication não retornou o UID da conta recriada.");
    }

    await firestorePatchDoc(env, token, "usuarios", whatsapp, {
      authUid: novoUid,
      email,
      atualizadoEm: Date.now(),
      contaAuthReparadaEm: Date.now(),
    });

    return {
      localId: novoUid,
      email,
      recriado: true,
    };
  }

  // Repara authUid antigo/desalinhado se localizou a conta pelo e-mail.
  if (text(authUser.localId) && text(authUser.localId) !== uidSalvo) {
    await firestorePatchDoc(env, token, "usuarios", whatsapp, {
      authUid: text(authUser.localId),
      atualizadoEm: Date.now(),
      contaAuthReparadaEm: Date.now(),
    });
  }

  return authUser;
}

function firestoreBase(env) {
  const projectId = encodeURIComponent(text(env.FIREBASE_PROJECT_ID));
  return `https://firestore.googleapis.com/v1/projects/${projectId}/databases/(default)/documents`;
}

function fromFirestoreValue(v) {
  if (!v || typeof v !== "object") return null;

  if ("nullValue" in v) return null;
  if ("stringValue" in v) return v.stringValue;
  if ("booleanValue" in v) return Boolean(v.booleanValue);
  if ("integerValue" in v) return Number(v.integerValue);
  if ("doubleValue" in v) return Number(v.doubleValue);
  if ("timestampValue" in v) return Date.parse(v.timestampValue);

  if ("arrayValue" in v) {
    return (v.arrayValue.values || []).map(fromFirestoreValue);
  }

  if ("mapValue" in v) {
    return fromFirestoreFields(v.mapValue.fields || {});
  }

  return null;
}

function fromFirestoreFields(fields = {}) {
  const out = {};
  for (const [key, value] of Object.entries(fields)) {
    out[key] = fromFirestoreValue(value);
  }
  return out;
}

function toFirestoreValue(value) {
  if (value === null || value === undefined) {
    return { nullValue: null };
  }

  if (typeof value === "boolean") {
    return { booleanValue: value };
  }

  if (typeof value === "number") {
    if (Number.isInteger(value)) return { integerValue: String(value) };
    return { doubleValue: value };
  }

  if (Array.isArray(value)) {
    return { arrayValue: { values: value.map(toFirestoreValue) } };
  }

  if (typeof value === "object") {
    const fields = {};
    for (const [k, v] of Object.entries(value)) {
      fields[k] = toFirestoreValue(v);
    }
    return { mapValue: { fields } };
  }

  return { stringValue: String(value) };
}

function fieldsPayload(obj) {
  const fields = {};
  for (const [key, value] of Object.entries(obj)) {
    fields[key] = toFirestoreValue(value);
  }
  return { fields };
}

async function firestoreGetDoc(env, token, collection, docId) {
  const url = `${firestoreBase(env)}/${encodeURIComponent(collection)}/${encodeURIComponent(docId)}`;

  const response = await fetch(url, {
    headers: { Authorization: `Bearer ${token}` },
  });

  if (response.status === 404) return null;

  const data = await response.json();

  if (!response.ok) {
    throw new Error(data?.error?.message || `Falha ao ler ${collection}/${docId}.`);
  }

  return {
    id: docId,
    ...fromFirestoreFields(data.fields || {}),
  };
}

async function firestoreListCollection(env, token, collection) {
  let pageToken = "";
  const docs = [];

  do {
    const params = new URLSearchParams({
      pageSize: "300",
    });

    if (pageToken) params.set("pageToken", pageToken);

    const url = `${firestoreBase(env)}/${encodeURIComponent(collection)}?${params.toString()}`;

    const response = await fetch(url, {
      headers: { Authorization: `Bearer ${token}` },
    });

    const data = await response.json();

    if (!response.ok) {
      throw new Error(data?.error?.message || `Falha ao listar ${collection}.`);
    }

    for (const doc of data.documents || []) {
      const id = decodeURIComponent(doc.name.split("/").pop());
      docs.push({
        id,
        ...fromFirestoreFields(doc.fields || {}),
      });
    }

    pageToken = data.nextPageToken || "";
  } while (pageToken);

  return docs;
}

async function firestorePatchDoc(env, token, collection, docId, patch) {
  const params = new URLSearchParams();

  for (const field of Object.keys(patch)) {
    params.append("updateMask.fieldPaths", field);
  }

  const url =
    `${firestoreBase(env)}/${encodeURIComponent(collection)}/${encodeURIComponent(docId)}` +
    `?${params.toString()}`;

  const response = await fetch(url, {
    method: "PATCH",
    headers: {
      Authorization: `Bearer ${token}`,
      "Content-Type": "application/json",
    },
    body: JSON.stringify(fieldsPayload(patch)),
  });

  const data = await response.json();

  if (!response.ok) {
    throw new Error(data?.error?.message || `Falha ao atualizar ${collection}/${docId}.`);
  }

  return {
    id: docId,
    ...fromFirestoreFields(data.fields || {}),
  };
}


async function firestorePatchAndDeleteFields(
  env,
  token,
  collection,
  docId,
  patch,
  deleteFields = []
) {
  const params = new URLSearchParams();

  for (const field of Object.keys(patch)) {
    params.append("updateMask.fieldPaths", field);
  }

  for (const field of deleteFields) {
    params.append("updateMask.fieldPaths", field);
  }

  const url =
    `${firestoreBase(env)}/${encodeURIComponent(collection)}/${encodeURIComponent(docId)}` +
    `?${params.toString()}`;

  const response = await fetch(url, {
    method: "PATCH",
    headers: {
      Authorization: `Bearer ${token}`,
      "Content-Type": "application/json",
    },
    body: JSON.stringify(fieldsPayload(patch)),
  });

  const data = await response.json();

  if (!response.ok) {
    throw new Error(
      data?.error?.message ||
      `Falha ao atualizar ${collection}/${docId}.`
    );
  }

  return {
    id: docId,
    ...fromFirestoreFields(data.fields || {}),
  };
}

async function firestoreDeleteDoc(env, token, collection, docId) {
  const url = `${firestoreBase(env)}/${encodeURIComponent(collection)}/${encodeURIComponent(docId)}`;

  const response = await fetch(url, {
    method: "DELETE",
    headers: {
      Authorization: `Bearer ${token}`,
    },
  });

  if (!response.ok && response.status !== 404) {
    let data = {};
    try { data = await response.json(); } catch (_) {}
    throw new Error(data?.error?.message || `Falha ao excluir ${collection}/${docId}.`);
  }

  return true;
}

function expiryMs(user) {
  const value = user?.expiraEm;

  if (Number.isFinite(Number(value)) && Number(value) > 0) {
    const n = Number(value);
    return n < 100000000000 ? n * 1000 : n;
  }

  const days = Number(user?.duracaoDias || 0);
  const start = Number(user?.ativadoEm || user?.criadoEm || 0);

  if (days > 0 && start > 0) {
    return start + days * DAY_MS;
  }

  return 0;
}

function isLifetime(user, now = Date.now()) {
  return (
    user?.vitalicio === true ||
    /vital[ií]cio/i.test(String(user?.plano || "")) ||
    Number(user?.duracaoDias || 0) >= LIFETIME_DAYS ||
    (expiryMs(user) - now) / DAY_MS >= LIFETIME_DAYS
  );
}

function planName(days) {
  if (days >= LIFETIME_DAYS) return "Vitalício";
  if (days === 30) return "Mensal";
  if (days === 90) return "Trimestral";
  if (days === 7) return "Semanal";
  return `${days} Dias (Liberado)`;
}

export default {
  async fetch(request, env) {
    const url = new URL(request.url);

    // Compatibilidade do novo Painel do Dono (V53+) diretamente com este Worker.
    if (url.pathname === "/session/login") url.pathname = "/admin/login";
    if (url.pathname.startsWith("/api/admin/")) {
      url.pathname = url.pathname.replace(/^\/api\/admin\//, "/admin/");
    }

    if (request.method === "OPTIONS") {
      return new Response(null, {
        status: 204,
        headers: CORS_HEADERS,
      });
    }

    if (url.pathname === "/" && request.method === "GET") {
      return new Response("API Pacote É Mato Online!", {
        status: 200,
        headers: {
          ...CORS_HEADERS,
          "Content-Type": "text/plain; charset=utf-8",
        },
      });
    }

    /* =====================================================
       VINCULAR CONTA ANTIGA
       Valida a sessão Firebase e a senha antiga no backend.
    ===================================================== */

    if (
      url.pathname === "/vincular-conta-antiga" &&
      request.method === "POST"
    ) {
      try {
        const body = await request.json();

        const whatsapp = digits(body.whatsapp);
        const senhaAntiga = String(body.senhaAntiga || "").trim();
        const idToken = text(body.firebaseIdToken);
        const apiKey = text(body.firebaseApiKey);
        const deviceId = text(body.deviceId);

        if (!whatsapp) {
          return json({ sucesso: false, erro: "WhatsApp inválido." }, 400);
        }

        if (!senhaAntiga) {

            pendente: true,
            status: payment.status || "pending",
            erro: "Pagamento ainda não aprovado.",
          }, 409);
        }

        const expectedReference = `pacote-emato-${whatsapp}`;

        if (String(payment.external_reference || "") !== expectedReference) {
          return json({
            sucesso: false,
            erro: "Este pagamento não pertence a esta conta.",
          }, 403);
        }

        /* 2. O plano é decidido pelo valor real recebido.
              Nunca confiamos nos dias enviados pelo navegador. */
        const amount = Number(payment.transaction_amount || 0);

        let dias = 0;
        let plano = "";
        let valorPlano = 0;

        if (Math.abs(amount - 4.99) < 0.001) {
          dias = 7;
          plano = "7 Dias (Semanal)";
          valorPlano = 4.99;
        } else if (Math.abs(amount - 9.99) < 0.001) {
          dias = 30;
          plano = "30 Dias (Mensal)";
          valorPlano = 9.99;
        } else if (Math.abs(amount - 24.99) < 0.001) {
          dias = 90;
          plano = "90 Dias (3 Meses)";
          valorPlano = 24.99;
        } else {
          return json({
            sucesso: false,
            erro: "O valor pago não corresponde a nenhum plano válido.",
          }, 400);
        }

        const googleToken = await getGoogleAccessToken(env);

        /* 3. Idempotência: se esse pagamento já foi aplicado, não aplica de novo */
        const pagamentoExistente = await firestoreGetDoc(
          env,
          googleToken,
          "pagamentos_concluidos",
          paymentId
        );

        if (pagamentoExistente) {
          return json({
            sucesso: true,
            jaProcessado: true,
            mensagem: "Pagamento já havia sido confirmado.",
            plano: pagamentoExistente.plano || plano,
            expiraEm: pagamentoExistente.expiraEm || 0,
          });
        }

        const user = await firestoreGetDoc(
          env,
          googleToken,
          "usuarios",
          whatsapp
        );

        if (!user) {
          return json({
            sucesso: false,
            erro: "Conta não encontrada para liberar o pagamento.",
          }, 404);
        }

        if (user.bloqueado === true) {
          return json({
            sucesso: false,
            erro: "A conta está bloqueada pelo administrador.",
          }, 403);
        }

        const now = Date.now();

        let bonusComprador = 0;
        let indicador = "";

        /* Usa primeiro o indicador já salvo na conta.
           Se ainda não houver, aceita o cupom informado nessa primeira compra. */
        if (
          user.bonusIndicacaoAplicado !== true &&
          cupomInformado &&
          cupomInformado !== whatsapp
        ) {
          indicador = digits(user.indicadoPor) || cupomInformado;

          const referrer = await firestoreGetDoc(
            env,
            googleToken,
            "usuarios",
            indicador
          );

          if (referrer) {
            bonusComprador = 3;

            const refBase = Math.max(now, expiryMs(referrer));

            await firestorePatchDoc(
              env,
              googleToken,
              "usuarios",
              indicador,
              {
                expiraEm: refBase + (7 * DAY_MS),
                ativo: true,
                totalIndicacoes: Number(referrer.totalIndicacoes || 0) + 1,
                atualizadoEm: now,
              }
            );
          } else {
            indicador = "";
          }
        }

        const totalDias = dias + bonusComprador;

        /* Mantém dias que ainda restavam quando a assinatura ainda está ativa */
        const baseExpira = Math.max(now, expiryMs(user));
        const newExpiry = baseExpira + (totalDias * DAY_MS);

        const patchUser = {
          whatsapp,
          expiraEm: newExpiry,
          ativo: true,
          bloqueado: false,
          plano: plano + (bonusComprador ? " (+3d Bônus Indicação)" : ""),
          duracaoDias: dias,
          vitalicio: false,
          tipo: "pago",
          atualizadoEm: now,
        };

        if (indicador) {
          patchUser.indicadoPor = indicador;
          patchUser.bonusIndicacaoAplicado = true;
        }

        await firestorePatchDoc(
          env,
          googleToken,
          "usuarios",
          whatsapp,
          patchUser
        );

        await firestorePatchDoc(
          env,
          googleToken,
          "pagamentos_concluidos",
          paymentId,
          {
            whatsapp,
            plano,
            valor: valorPlano,
            data: new Date(now).toLocaleString("pt-BR", {
              timeZone: "America/Sao_Paulo",
            }),
            timestamp: now,
            expiraEm: newExpiry,
            mercadoPagoStatus: payment.status,
            mercadoPagoId: String(payment.id || paymentId),
          }
        );

        return json({
          sucesso: true,
          mensagem: "Pagamento confirmado e acesso liberado.",
          plano: patchUser.plano,
          dias,
          bonusDias: bonusComprador,
          expiraEm: newExpiry,
        });

      } catch (e) {
        console.error("ERRO FINALIZAR PIX:", e);

        return json({
          sucesso: false,
          erro: e?.message || "Erro interno ao finalizar pagamento.",
        }, 500);
      }
    }

    /* =====================================================
       PIX
    ===================================================== */

    if (url.pathname === "/criar-pix" && request.method === "POST") {
      try {
        const body = await request.json();

        const token = text(env.MP_ACCESS_TOKEN);
        const value = Number(body.valor);
        const whatsapp = digits(body.zap);

        if (!token) {
          return json({ erro: "MP_ACCESS_TOKEN não configurado." }, 500);
        }

        if (!Number.isFinite(value) || value <= 0) {
          return json({ erro: "Valor do pagamento inválido." }, 400);
        }

        if (!whatsapp) {
          return json({ erro: "WhatsApp inválido." }, 400);
        }

        const payerEmail =
          isEmail(body.email)
            ? String(body.email).trim().toLowerCase()
            : `cliente_${whatsapp}_${Date.now()}@pacoteemato.com`;

        const response = await fetch(
          "https://api.mercadopago.com/v1/payments",
          {
            method: "POST",
            headers: {
              Authorization: `Bearer ${token}`,
              "Content-Type": "application/json",
              "X-Idempotency-Key": `pacote-emato-${whatsapp}-${Date.now()}`,
            },
            body: JSON.stringify({
              transaction_amount: value,
              description: text(body.descricao) || "Pacote É Mato - Acesso",
              payment_method_id: "pix",
              payer: {
                email: payerEmail,
                first_name: text(body.nome) || "Cliente",
                last_name: "PacoteEmato",
              },
              external_reference: `pacote-emato-${whatsapp}`,
            }),
          }
        );

        let data;

        try {
          data = await response.json();
        } catch (_) {
          return json({
            erro: "Resposta inválida do Mercado Pago.",
            status_http: response.status,
          }, 502);
        }

        if (!response.ok) {
          return json({
            erro:
              data?.cause?.[0]?.description ||
              data?.message ||
              "Mercado Pago recusou a criação do Pix.",
            detalhe: data,
          }, response.status);
        }

        return json(data, 200);
      } catch (e) {
        return json({
          erro: e?.message || "Erro interno ao criar Pix.",
        }, 500);
      }
    }

    if (url.pathname.startsWith("/status-pix/") && request.method === "GET") {
      try {
        const token = text(env.MP_ACCESS_TOKEN);
        const paymentId = url.pathname.split("/")[2];

        if (!token) {
          return json({ erro: "MP_ACCESS_TOKEN não configurado." }, 500);
        }

        if (!paymentId) {
          return json({ erro: "ID do pagamento não informado." }, 400);
        }

        const response = await fetch(
          `https://api.mercadopago.com/v1/payments/${encodeURIComponent(paymentId)}`,
          {
            headers: {
              Authorization: `Bearer ${token}`,
            },
          }
        );

        const data = await response.json();

        if (!response.ok) {
          return json({
            erro: data?.message || "Erro ao consultar Pix.",
            detalhe: data,
          }, response.status);
        }

        return json(data, 200);
      } catch (e) {
        return json({
          erro: e?.message || "Erro interno ao consultar Pix.",
        }, 500);
      }
    }

    return json({ erro: "Endpoint não encontrado." }, 404);
  },
};
